CREATE SCHEMA IF NOT EXISTS abg;

CREATE TABLE abg.users (
 id SERIAL NOT NULL,
 row_id TEXT NOT NULL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(255) NOT NULL,
    password_hash TEXT NOT NULL,
    role VARCHAR(20) NOT NULL DEFAULT 'USER'
        CHECK (role IN ('USER', 'ADMIN')),
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
     cr_on timestamp without time zone DEFAULT now() NOT NULL,
 up_on timestamp without time zone DEFAULT now() NOT NULL
	);



CREATE UNIQUE INDEX uq_abg_users_email
ON abg.users (LOWER(email));


ALTER TABLE abg.users
ADD COLUMN IF NOT EXISTS deleted_at TIMESTAMP WITHOUT TIME ZONE
DEFAULT NULL;



ALTER TABLE abg.users
ALTER COLUMN is_active SET DEFAULT FALSE;
    
CREATE TABLE IF NOT EXISTS abg.patients (
    id BIGSERIAL PRIMARY KEY,

    row_id VARCHAR(50) UNIQUE NOT NULL,

    name VARCHAR(100) NOT NULL,

    age SMALLINT NOT NULL
        CHECK (age BETWEEN 0 AND 120),

    gender VARCHAR(20) NOT NULL
        CHECK (gender IN ('Male', 'Female', 'Other')),

    ipd_no VARCHAR(50) NOT NULL,

    bed_no VARCHAR(50) NOT NULL,

    diagnosis TEXT NOT NULL,

    old_history TEXT,

    created_by VARCHAR(50),

    cr_on TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    up_on TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX IF NOT EXISTS idx_patients_ipd_no
    ON abg.patients (ipd_no);

CREATE INDEX IF NOT EXISTS idx_patients_name
    ON abg.patients (name);

CREATE INDEX IF NOT EXISTS idx_patients_created_by
    ON abg.patients (created_by);



CREATE TABLE IF NOT EXISTS abg.patient_sugar (
id BIGSERIAL PRIMARY KEY,

    row_id VARCHAR(50) UNIQUE NOT NULL,
    patient_id BIGINT NOT NULL
        REFERENCES abg.patients(id) ON DELETE CASCADE,
    sugar NUMERIC(6,2) NOT NULL CHECK (sugar > 0 AND sugar <= 2000),
    treatments TEXT,
cr_on TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    up_on TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX IF NOT EXISTS idx_patient_sugar_patient_id
ON abg.patient_sugar(patient_id);
